# Skinned model header (actor models: player, monkeys, gadgets). Data follows the 0xC-byte
# header: colors[colorCount] (u32), bones[boneCount] (12 bytes), verts[vertCount], prims.
struct ModelHeader 0xC
0x0 u32       flags
0x4 u16       boneCount
0x6 u16       vertCount
0x8 u16       colorCount
0xA u16       hasNormals       ; nonzero: 16-byte verts with normals, drawn environment-mapped
end

# Player object (single global). Layout recovered from absolute references
# (tools/ghidra_scripts/DumpDataRefs.java -> notes/inventory/player-refs.csv).
# Shares the object header pattern with Monkey and the RC car: +0x18 anim set, +0x24 pos.
struct Player 0x150
# Common actor/model-instance header (same layout in Monkey; see ModelInit / ModelPoseBones).
0x000 ModelHeader* model
0x004 u32     colors           ; vertex colors (CVECTOR[colorCount])
0x008 u32     prims            ; primitive list (GPU-code runs, see ModelEmitPrims)
0x00C u32     bones            ; 12 bytes per bone {vertCount u16, firstVert u16, offset s16[3], parent u16}
0x010 u32     verts            ; SVECTOR[vertCount] (16 bytes each when the model has normals)
0x014 MATRIX* boneMatrices     ; [0] root, [1..] bones (shared buffer 0x800E9C60)
0x018 u32     animSet          ; animation frames: (boneCount + 2) words per frame, 10-bit packed ZYX angles
0x01C s16[3]  posShort         ; short copy of the position (x, ground y, z): set at spawn, read by PlayerAnimateAndDraw / Gadget4TrailUpdate (Tentative)
0x024 s32[3]  pos              ; world position; 139 functions read it
0x030 s32[3]  vel              ; added to pos when state == 0x2D
0x03C SVECTOR prevFacing       ; facing at the end of the last update
0x044 s16     bodyRadius       ; collision body radius (CollisionBody +0x20; 0x40 at spawn, also used by PlayerTryLedgeGrab)
0x046 s16     bodyHeight       ; collision body height (CollisionBody +0x22; 0x100 at init)
0x050 u32     contact          ; what the player touched: >>28 = type (4/5 push-off, 0xA wall, 0xB hazard), &0xFFF = object index
0x058 s16     contactCollider  ; collider index recorded by BodyRecordColliderContact (CollisionBody +0x34)
0x05C u16     animFrame        ; current animation frame; runs between the state's [first,last] in the state table
0x05E u16     state            ; index into the state table (>= 0x80 = gadget-specific table); set by PlayerSetState
0x060 SVECTOR facing           ; facing direction (x,y,z,pad), 0x1000 = 1.0
0x068 s16[3]  upVector         ; model tilt; follows the slope / right stick per the state's tilt mode
0x070 u32     stateFlags       ; flag nibbles copied from the state table entry by PlayerSetState (hi nibble 0xC/0xD/0xE tested)
0x074 SVECTOR desiredDir       ; direction to turn towards (TurnTowardsY), usually camera-relative stick
0x07C s32     pitch            ; angle for turn mode 4 (swimming up/down)
0x080 SVECTOR savedFacing      ; facing saved on water entry
0x088 s16     nextFrame
0x08A u16     animLerp         ; 0..0xFFF fraction between animFrame and nextFrame (AnimDecodeLerp t)
0x08C u16     animBlend        ; 0 = none, 5 = blend frames when changing state, 0xFFFE/0xFFFF special
0x08E u16     animSpeed        ; 0x1000 normal (PlayerSetState), 0x800 half (transition actions)
0x090 u32     animBlendFrom
0x0B4 s16[6]  climbTarget      ; climb attach point (x,y,z, ?, ?, height); set by PlayerSetClimbTarget, used by PlayerUpdateClimb
0x0C0 s16[8]  rope             ; two end points [0..2] [3..5] + position along it [6] (0..0x1000); PlayerUpdateRope
0x0D8 s16[3]  attachDir        ; facing / push direction stored by PlayerAttachSet (x, y, z)
0x0DE u16     attachTimer      ; frames left (PlayerAttachSet arg 2, PlayerCond_AttachTimerDone)
0x0E0 u32     attachFlags      ; bit1 climb target set, bit2 rope set, bit3 0xD8 attach set
0x0E4 u32     prevAttachFlags  ; attachFlags of the previous frame
0x0E8 s16     health           ; input ignored when 0; blink code checks < 1 (name tentative)
0x0EA s16     invulnTimer      ; 0x3C after PlayerDamage, counted down in PlayerOnContact
0x0EC s16     surfaceKind      ; index into the surface table 0x800AEF60; -1 airborne
0x0EE u8      unk_ee
0x0EF u8      contactFrames    ; consecutive contact frames (PlayerOnContact acts after 5)
0x0F0 u8      hurt             ; 1 after PlayerDamage, cleared by PlayerOnContact
0x0F1 u8      contactCooldown  ; 0xF, counted down by PlayerOnContact
0x0F2 s8      currentGadget    ; 7 = RC car
0x0F3 s8      savedGadget      ; restored into currentGadget on room change (StageRequest code 6)
0x0F4 u8[4]   tint             ; current model color (r,g,b); fades toward tintTarget
0x0F8 u8[4]   tintTarget
0x0FC s16     fallFrames       ; frames at terminal fall speed; PlayerAct_Land hurts when >= 6
0x0FE s16     squashTimer      ; >0 scales the model (squash/stretch)
0x100 u16     squashFrame      ; frame counter for PlayerDrawSquashed
0x102 s16     unk_102
0x104 s16     slopeX           ; ground slope (x); slides / limits per surface table
0x106 s16     slopeZ
0x108 s32[3]  velFrac          ; sub-unit remainder of moveVel after vel = moveVel >> 12 (also external push)
0x114 s32[3]  targetMoveVel    ; desired velocity; PlayerAccelXZ/XYZ move moveVel toward it
0x120 s32[3]  moveVel          ; movement velocity (y negative = up; jumps set -0x1E000..-0x80000)
0x12C s32     groundY          ; floor height under the player (shadow height)
0x130 s16[3]  lastPos          ; compared with pos to detect idling
0x136 s16     hasGround        ; floor found under the player
0x138 u16     groundType       ; >>12 == 1: water (then waterY is the surface)
0x13A u16     prevGroundType   ; groundType of the previous frame
0x13C s16     waterY
0x13E u16     unk_13e
0x148 u16     idleTimer        ; idle animation streams in after 900 frames
0x14A u8      idleAnimLoading
0x14B u8      idleAnimReady
end

# One state-table entry (decoded from PlayerUpdate / PlayerSetState).
struct PlayerStateEntry 0x8
0x0 u16       firstFrame
0x2 u16       lastFrame
0x4 u8        turnTilt          ; lo nibble: turn mode (how facing follows input); hi nibble: model tilt mode
0x5 u8        moveMode          ; how moveVel is derived (0..0x10, switch in PlayerUpdate)
0x6 u8        frameAction       ; g_PlayerActionFns index called every frame in this state (0 = none)
0x7 u8        flags             ; copied to Player.stateFlags (hi nibble = state class: 0x4 air, 0xA wall, 0xC/0xE water...)
end

# One per gadget (+1; index 0 = no gadget). Indexed as [currentGadget + 1]. Gadget 9 = none.
struct PlayerStateSet 0x10
0x00 u32      animSet
0x04 PlayerStateEntry* stateTable
0x08 u32      transitions       ; PlayerRunTransitions table
0x0C u32      events            ; PlayerRunAnimEvents list
end

# Scratchpad block player code works from. PlayerUpdate copies pad/stick state into it
# each frame (zeroed when input is locked). Left stick moves, right stick swings gadgets
# (raw DualShock order RX RY LX LY; movement uses left, flicks/rotations come from right).
struct PlayerScratch 0x3C
0x00 PlayerStateEntry* stateTable ; active PlayerStateSet fields for the current state range
0x04 u32      transitions
0x08 u32      events
0x0C u32      held             ; buttons held (PsyQ bits: 0x0A = R1|R2 jump, 0x200 L3, 0x400 R3)
0x10 u32      changed          ; buttons changed this frame
0x14 u16      leftStickClass      ; 0 / 2 / 4 / 8 by magnitude (copy of 0x800E5498)
0x16 u16      rightStickGesture    ; 4/8 flick, 0x10/0x20 rotation, 0x80 (copy of 0x800E549A)
0x18 u16      unk_18
0x1A u16      gestureEvents    ; one-shot gesture flags (copy of 0x800E549E)
0x1C u16      leftStickMag        ; 0..0x7F
0x1E u16      rightStickMag        ; 0..0x7F
0x20 s32      unk_20
0x24 s32      unk_24
0x28 s16[2]   rightStickCam       ; right stick relative to camera (fwd/side, +-0x1000)
0x2C s16[2]   leftStickCam        ; left stick relative to camera
0x30 s16      stickRotation    ; signed rotation amount
0x32 s16      unk_32
0x34 u16      stickTravelR     ; summed stick movement over the history
0x36 u16      stickTravelL
0x38 u16      inputLocked      ; nonzero: player input ignored
end

# Terrain heightfield + render settings for the current area (filled by AreaLoadGridAndProps /
# AreaLoad). Cells are 0x200 units; the grid is centered on the origin.
struct LevelGrid 0x30
0x00 u16      width            ; cells in x
0x02 u16      depth            ; cells in z
0x04 u32      colorCount
0x08 u32      cells            ; u32[width*depth]: hi16 surface attribute, bit 11 + bits 8-10 triangle split
0x0C u32      colors           ; CVECTOR[colorCount] terrain vertex colors
0x10 u32      cellTiles        ; 8 bytes per cell: corner color indices u8[4] + tile/texture info u8[4]
0x14 u32      heights          ; s16[(w+1)*(d+1)] corner heights, then s16[w*d] cell-center heights
0x18 u32      blockMask        ; u8 per 2x2 block, -1 = nothing to draw (BuildTerrainBlockMask)
0x1C u8[4]    fogColor         ; far color (GTE FC) from area header +0x20
0x20 u32      unk_20
0x24 u32      fogParam         ; 0xF000
0x28 u32      fogStart         ; from area header +0x2C (default 0x4B0)
0x2C u32      drawDistance     ; fogStart * 4 + 0x400
end

# Collider box (0x38 bytes; table pointer g_Colliders, count g_ColliderCount).
struct Collider 0x38
0x00 s16[3]   pos              ; center x, top y, z
0x06 s16[3]   halfSize         ; half extents x, height, z
0x0E s16      propIndex        ; prop (g_AreaProps) moved / re-meshed with this collider, -1 none
0x10 u16      geometry         ; geometry / response handler word (dispatch class in the high nibble, shape type; codex notes/69)
0x12 u16      type             ; surface / contact type: low byte kind (9 = water volume); bits 8-11 surface nibble
0x14 u16      mode             ; bit 15 = solid mode (ColliderTypeActive; codex notes/69)
0x18 u16      flags            ; 0x8000 = disabled
0x1E s16      touchedBy        ; reset to -1 each frame by CollisionUpdate
0x20 u16      zoneMask         ; quadrant mask (ColliderComputeQuadrantMask), tested against GetColliderQuadrantMaskXZ
0x24 s16[3]   origin           ; closed / pivot position used by door gimmicks
0x2A s16[3]   flow             ; current / push vector (x, y, z) << 12 in GetSurfaceType
end

# Interactive area object ("gimmick"): switches, doors, breakables, triggers (0x18 bytes;
# array g_Gimmicks from the area blob +0x4D4, grouped by g_GimmickGroups).
struct Gimmick 0x18
0x00 u16      type
0x02 u16      state            ; doors: 0 closed, 2 opening, 3 open, 5 closing (1/4 = start states)
0x04 s16      collider         ; g_Colliders index (its propIndex is the visual)
0x06 s16      timer            ; door travel / hit counter
0x08 s16      subtype          ; e.g. door direction 0-7, switch kind
0x0A s16      param_a
0x0C s16      link             ; gimmick activated by this one (GimmickActivate), -1 none
0x0E s16      param_e          ; trigger radius for proximity triggers
0x10 s16      param_10
0x12 s16      param_12         ; unique id for proximity triggers (progress flag)
0x14 s16      param_14
0x16 s16      param_16
end

# Area header inside the area data blob (0x80154000 + [0x801544D0]); offsets are in words.
struct AreaHeader 0x54

0x04 u32      gridOfs          ; -> LevelGrid data
0x08 u32      meshTableOfs     ; -> mesh table (g_AreaMeshTable)
0x0C u32      propsOfs         ; -> count, then 12-byte placements {mesh u16, rotY u16, x s16, y s16, z s16, ?}
0x10 u32      collisionOfs     ; -> {shapeGridOfs, shapesOfs, collidersOfs, list3Ofs, list4Ofs} (byte offsets)
0x20 u8[4]    fogColor
0x24 u32      unk_24
0x28 s16      fogEnabled
0x2A s16      unk_2a
0x2C u16      fogNear
0x2E u16      fogFar
0x34 u32[8]   features         ; low byte = feature id (0x08 0x10 0x11 0x20 0x30-0x32 0x40 0x41), upper bits = data offset
end
